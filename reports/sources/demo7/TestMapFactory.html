12 /**
13 * Locates and instantiates the {@link LRUClockMap} implementation under test.
14 *
15 * <p>Because the tests are written against the interface only and the concrete
16 * implementation is supplied later, this factory discovers the implementation
17 * at runtime using reflection. The lookup order is:
18 *
19 * <ol>
20 * <li>the class named by the system property {@code lruclockmap.impl}</li>
21 * <li>a {@code public static of(int)} factory method on {@link LRUClockMap} itself
22 * (the interface publishes this as implementation guidance)</li>
23 * <li>the conventional name {@code org.jugsaxony.clock.demo7.LRUClockMapImpl}</li>
24 * <li>a best-effort scan of the {@code org.jugsaxony.clock.demo7} package on the
25 * classpath for a concrete class implementing {@link LRUClockMap} that offers
26 * either a {@code static of(int)} method or an {@code (int)} constructor</li>
27 * </ol>
28 *
29 * <p>Exceptions thrown by the implementation factory (for example the
30 * {@link IllegalArgumentException} for a maxSize below 4) are propagated unchanged.
31 */
32 final class TestMapFactory
33 {
34 /** System property to explicitly name the implementation class. */
35 static final String IMPL_PROPERTY = "lruclockmap.impl";
36
37 private static final String PACKAGE_NAME = "org.jugsaxony.clock.demo7";
38 private static final String PACKAGE_PATH = PACKAGE_NAME.replace('.', '/');
39
40 private static volatile Creator cachedCreator;
41
42 private TestMapFactory()
43 {
44 // utility class
45 }

47 @FunctionalInterface
48 private interface Creator
49 {
50 Object create(int maxSize) throws Exception;
51 }
52
53 /**
54 * Creates a new map instance from the discovered implementation.
55 *
56 * @param maxSize the maximum size of the map
57 * @return a new LRUClockMap
58 * @throws IllegalArgumentException if the implementation rejects the maxSize
59 * @throws IllegalStateException if no implementation can be found
60 */
61 @SuppressWarnings("unchecked")
62 static <K, V> LRUClockMap<K, V> create(final int maxSize)
63 {
64 var creator = cachedCreator;
65 if (creator == null)
66 {
67 creator = resolve();
68 cachedCreator = creator;
69 }
70
71 try
72 {
73 return (LRUClockMap<K, V>) creator.create(maxSize);
74 }
75 catch (final InvocationTargetException e)
76 {
77 // propagate what the implementation threw, e.g. IllegalArgumentException
78 switch (e.getCause())
79 {
80 case RuntimeException re -> throw re;
81 case Error err -> throw err;
82 case Throwable t -> throw new IllegalStateException(
83 "LRUClockMap factory failed for maxSize " + maxSize, t);
84 case null -> throw new IllegalStateException(
85 "LRUClockMap factory failed for maxSize " + maxSize, e);
86 }
87 }
88 catch (final RuntimeException e)
89 {
90 throw e;
91 }
92 catch (final Exception e)
93 {
94 throw new IllegalStateException(
95 "Could not create LRUClockMap for maxSize " + maxSize, e);
96 }
97 }

99 private static Creator resolve()
100 {
101 var candidates = new ArrayList<String>();
102
103 var fromProperty = System.getProperty(IMPL_PROPERTY);
104 if (fromProperty != null && !fromProperty.isBlank())
105 {
106 candidates.add(fromProperty.trim());
107 }
108 candidates.add(LRUClockMap.class.getName());
109 candidates.add(PACKAGE_NAME + ".LRUClockMapImpl");
110 candidates.addAll(scanPackage());
111
112 var problems = new ArrayList<String>();
113 for (final var candidate : candidates)
114 {
115 try
116 {
117 var type = Class.forName(candidate);
118 var creator = creatorFor(type);
119 if (creator != null)
120 {
121 return creator;
122 }
123 problems.add(candidate + ": no usable static of(int) or (int) constructor");
124 }
125 catch (final ClassNotFoundException | NoClassDefFoundError e)
126 {
127 problems.add(candidate + ": class not found");
128 }
129 }
130
131 throw new IllegalStateException("""
132 No LRUClockMap implementation found. Tried:
133 %s
134 Provide one via the system property '%s', a static of(int) on the \
135 interface, or a concrete implementation class in package %s."""
136 .formatted(String.join(System.lineSeparator() + " ", problems),
137 IMPL_PROPERTY, PACKAGE_NAME));
138 }

140 /**
141 * Builds a creator for a candidate type or returns null if the type
142 * cannot serve as an implementation.
143 */
144 private static Creator creatorFor(final Class<?> type)
145 {
146 // preferred: public static of(int)
147 try
148 {
149 final Method of = type.getMethod("of", int.class);
150 final boolean isStatic = Modifier.isStatic(of.getModifiers());
151 final boolean returnsMap = LRUClockMap.class.isAssignableFrom(of.getReturnType());
152
153 if (isStatic && returnsMap)
154 {
155 of.setAccessible(true);
156 return maxSize -> of.invoke(null, maxSize);
157 }
158 }
159 catch (final NoSuchMethodException e)
160 {
161 // fall through to constructor lookup
162 }
163
164 // fallback: public (int) constructor on a concrete implementation class
165 final boolean isCandidateImpl = LRUClockMap.class.isAssignableFrom(type)
166 && !type.isInterface()
167 && !Modifier.isAbstract(type.getModifiers());
168
169 if (isCandidateImpl)
170 {
171 try
172 {
173 final var ctor = type.getConstructor(int.class);
174 return maxSize -> ctor.newInstance(maxSize);
175 }
176 catch (final NoSuchMethodException e)
177 {
178 // not usable
179 }
180 }
181
182 return null;
183 }

185 /**
186 * Best-effort scan of the package directory on the classpath to discover
187 * implementation class names. Works for directory-based classpaths, which
188 * is the normal case for locally built Maven modules.
189 */
190 private static List<String> scanPackage()
191 {
192 final var result = new ArrayList<String>();
193
194 try
195 {
196 final var loader = TestMapFactory.class.getClassLoader();
197 final Enumeration<URL> urls = loader.getResources(PACKAGE_PATH);
198
199 while (urls.hasMoreElements())
200 {
201 final var url = urls.nextElement();
202 if (!"file".equals(url.getProtocol()))
203 {
204 continue;
205 }
206
207 final var dir = new File(url.toURI());
208 final var files = dir.listFiles((d, name) ->
209 name.endsWith(".class") && !name.contains("$"));
210 if (files == null)
211 {
212 continue;
213 }
214
215 for (final var file : files)
216 {
217 final var className = PACKAGE_NAME + "."
218 + file.getName().substring(0, file.getName().length() - ".class".length());
219 if (!className.equals(TestMapFactory.class.getName())
220 && !className.equals(LRUClockMap.class.getName()))
221 {
222 result.add(className);
223 }
224 }
225 }
226 }
227 catch (final Exception e)
228 {
229 // best effort only, the explicit candidates still apply
230 }
231
232 return result;
233 }
