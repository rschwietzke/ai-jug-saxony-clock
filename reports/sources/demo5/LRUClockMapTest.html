24 /**
25 * Black-box tests for {@link LRUClockMap}.
26 *
27 * <p>The suite only relies on the contract documented by the interface. The
28 * single creation point is {@link #newMap(int)}, which is expected to call the
29 * documented static factory {@code LRUClockMap.of(maxSize)} once the interface
30 * is backed by a real implementation.</p>
31 *
32 * <p>The clock eviction policy is only "good enough LRU"; therefore tests never
33 * hard-code which key is evicted. They assert the invariants that every correct
34 * clock implementation must uphold. The one semantic guarantee that is tested
35 * is the second chance: a key read via {@link LRUClockMap#get(Object)} must not
36 * be evicted by the next single eviction while untouched entries exist.</p>
37 */

40
41 /** The only creation point of the tests, matches the documented factory. */
42 private static <K, V> LRUClockMap<K, V> newMap(final int maxSize) {
43 return LRUClockMap.of(maxSize);
44 }

593 @Nested
594 @DisplayName("randomized invariants")
595 class RandomizedInvariants {
596
597 @Test
598 @DisplayName("mixed put/get/remove keeps size, keys and values consistent")
599 void mixedOperationsPreserveInvariants() {
600 final int maxSize = 8;
601 final LRUClockMap<Integer, Integer> map = newMap(maxSize);
602 final Random random = new Random(42);
603 final Set<Integer> live = new HashSet<>();
604 int valueCounter = 0;
605
606 for (int i = 0; i < 5_000; i++) {
607 switch (random.nextInt(4)) {
608 case 0, 1 -> {
609 final int key = random.nextInt(50);
610 final int value = ++valueCounter;
611 final boolean wasLive = live.contains(key);
612
613 map.put(key, value);
614
615 if (!wasLive && live.size() == maxSize) {
616 live.removeIf(k -> map.getRaw(k) == null);
617 assertEquals(maxSize - 1, live.size(),
618 "a new key inserted while full must evict exactly one key");
619 }
620 live.add(key);
621 assertEquals(value, map.getRaw(key));
622 }
623 case 2 -> {
624 final int key = random.nextInt(50);
625 final Integer expected = live.contains(key) ? map.getRaw(key) : null;
626 assertEquals(expected, map.get(key));
627 }
628 default -> {
629 final int key = random.nextInt(50);
630 final boolean wasLive = live.remove(key);
631 final Integer removed = map.remove(key);
632 if (wasLive) {
633 assertNotNull(removed, "removing a live key must return its value");
634 } else {
635 assertNull(removed, "removing an absent key must return null");
636 }
637 }
638 }
639
640 assertInvariants(map, maxSize, live);
641 }
642
643 map.clear();
644
645 assertAll(
646 () -> assertEquals(0, map.size()),
647 () -> assertTrue(map.keys().isEmpty()));
648 }
649
650 private static void assertInvariants(final LRUClockMap<Integer, Integer> map,
651 final int maxSize,
652 final Set<Integer> live) {
653 final List<Integer> keys = map.keys();
654 final Set<Integer> keySet = new HashSet<>(keys);
655
656 assertTrue(map.size() <= maxSize, "size must stay bounded");
657 assertEquals(map.size(), keys.size(), "size() must match keys().size()");
658 assertEquals(live, keySet, "keys() must match the tracked live keys");
659 assertFalse(keys.contains(null), "keys() must not contain null");
660 assertEquals(keys.size(), keySet.size(), "keys() must not contain duplicates");
661 keys.forEach(k -> assertNotNull(map.getRaw(k), "listed keys must be readable"));
662 }
663 }
