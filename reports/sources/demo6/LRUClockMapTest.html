28 /**
29 * Contract tests for {@link LRUClockMap}.
30 *
31 * <p>Everything here is written against the interface only. The tests never look
32 * at the internal array, the clock hand or the second chance flags, because none
33 * of that is visible. All expectations are derived from the six published methods
34 * and from the behaviour the Javadoc promises.
35 *
36 * <h2>Assumptions</h2>
37 * <ol>
38 * <li><b>Construction.</b> The Javadoc names {@code LRUClockMap.of(int)} as
39 * "the only way to create a new LRUClockMap". Every test goes through
40 * {@link #newMap(int)}, so if the implementation publishes construction in
41 * another way, that one method is the only thing that has to be adapted.</li>
42 * <li><b>Null hostility.</b> "No support for null keys or null values" is read
43 * as the idiomatic contract of a null hostile map (see {@code Map.of} or
44 * {@code ConcurrentHashMap}): a {@link NullPointerException}, not silent
45 * corruption. Those tests sit in {@link NullHandling} alone so they are
46 * easy to relax.</li>
47 * <li><b>Clock semantics.</b> The eviction tests assume a textbook second
48 * chance clock: a hand walks the occupied slots in a fixed order, clears a
49 * set flag and moves on, and evicts at the first entry whose flag is
50 * clear. Nothing assumes where the hand starts or in which order the slots
51 * are visited. The tests build the same map twice from an identical
52 * operation script, do one single extra thing to the second copy, and only
53 * compare the difference.</li>
54 * <li><b>Deterministic hashing.</b> Two maps built from the same operations
55 * end up in the same internal state, so the implementation must not mix a
56 * per instance random seed into the hash.</li>
57 * </ol>
58 *
59 * <p>JUnit 5 only, no other test or assertion library. Java 21 language level.
60 *
61 * <p>The whole suite runs in well under a second against a working
62 * implementation. The generous class level timeout is only there so that an
63 * implementation which loops forever - an easy mistake in the clock sweep and
64 * in a linear probe - reports a failure instead of hanging the build.
65 *
66 * @author generated contract test suite
67 */

143 /**
144 * How many over capacity inserts {@link #warmedMap(int)} performs after the
145 * initial fill. Deliberately smaller than {@code maxSize} so that the map
146 * afterwards holds a mix of old and freshly inserted entries, and therefore
147 * a mix of set and cleared second chance flags - no matter whether the
148 * implementation marks a freshly inserted entry as referenced or not.
149 *
150 * @param maxSize the maximum size of the map
151 * @return the number of warm up inserts
152 */
153 private static int warmUpCount(final int maxSize)
154 {
155 return Math.max(2, maxSize / 2);
156 }
157
158 /**
159 * Builds a map in a completely reproducible way: fill it to {@code maxSize},
160 * then perform a few more inserts so the clock hand has left its initial
161 * position. Two calls with the same {@code maxSize} execute exactly the same
162 * operations in exactly the same order and must therefore end up in exactly
163 * the same internal state.
164 *
165 * <p>This is the basis of all differential tests below: build the same map
166 * twice, do one single extra thing to the second copy, and see whether the
167 * next eviction changes.
168 *
169 * @param maxSize the maximum size of the map
170 * @return a filled and warmed up map
171 */
172 private static LRUClockMap<String, String> warmedMap(final int maxSize)
173 {
174 final var map = LRUClockMapTest.<String, String>newMap(maxSize);
175
176 fill(map, 0, maxSize);
177 fill(map, maxSize, warmUpCount(maxSize));
178
179 return map;
180 }
181
182 /**
183 * Determines which key a {@link #warmedMap(int)} gives up when
184 * {@link #INTRUDER} is inserted next.
185 *
186 * <p>Two throw away maps are used so that the map is never observed between
187 * the warm up and the insert. Otherwise the measurement itself could
188 * influence the very result it tries to measure.
189 *
190 * @param maxSize the maximum size of the map
191 * @return the key that gets evicted
192 */
193 private static String nextVictim(final int maxSize)
194 {
195 // first map: only ever read, never written after the warm up
196 final var before = keySet(warmedMap(maxSize));
197
198 // second map: written first, read afterwards
199 final var probe = warmedMap(maxSize);
200 probe.put(INTRUDER, "x");
201 final var after = keySet(probe);
202
203 assertEquals(maxSize, before.size(), "a warmed map must hold exactly maxSize entries");
204 assertTrue(after.contains(INTRUDER), "the newly inserted key must never be the one evicted");
205
206 final var evicted = new LinkedHashSet<>(before);
207 evicted.removeAll(after);
208
209 assertEquals(1, evicted.size(),
210 () -> "an over capacity insert must evict exactly one entry, but these went away: " + evicted);
211
212 return evicted.iterator().next();
213 }

1225 // ==================================================================
1226 // randomized model based test
1227 // ==================================================================
1228
1229 /**
1230 * Applies a long, seeded and therefore reproducible sequence of operations
1231 * and checks after every single one that everything the interface promises
1232 * still holds. The model cannot predict which entry the clock evicts, so it
1233 * only asserts what is knowable: values are never wrong, removed keys never
1234 * come back, the size never exceeds the limit, and keys() always agrees with
1235 * the rest of the map.
1236 */
1237 @Nested
1238 @DisplayName("randomized operations")
1239 class Randomized
1240 {
1241 /** The operations the generator can produce. */
1242 private sealed interface Op
1243 {
1244 /**
1245 * @param key the key to insert
1246 * @param value the value to insert
1247 */
1248 record Put(String key, String value) implements Op { }
1249
1250 /** @param key the key to read with an LRU effect */
1251 record Get(String key) implements Op { }
1252
1253 /** @param key the key to read without an LRU effect */
1254 record GetRaw(String key) implements Op { }
1255
1256 /** @param key the key to delete */
1257 record Remove(String key) implements Op { }
1258
1259 /** Drops everything. */
1260 record Clear() implements Op { }
1261 }
1262
1263 @ParameterizedTest(name = "seed {0}")
1264 @ValueSource(longs = { 1L, 7L, 42L, 2_026L, 987_654_321L, -5L })
1265 @DisplayName("all invariants hold through a long random operation sequence")
1266 void randomOperationsKeepTheMapConsistent(final long seed)
1267 {
1268 final var random = new Random(seed);
1269 final int maxSize = 4 + random.nextInt(40);
1270 final int keySpace = 3 * maxSize;
1271
1272 final var map = LRUClockMapTest.<String, String>newMap(maxSize);
1273
1274 // the value a key must carry if it is present at all
1275 final var expectedValues = new HashMap<String, String>();
1276 // keys that must not be in the map because they were removed or never added
1277 final var mustBeAbsent = new HashSet<String>();
1278
1279 for (int i = 0; i < keySpace; i++)
1280 {
1281 mustBeAbsent.add(key(i));
1282 }
1283
1284 for (int step = 0; step < 2_000; step++)
1285 {
1286 switch (nextOperation(random, keySpace, step))
1287 {
1288 case Op.Put(String k, String v) ->
1289 {
1290 final var previous = map.getRaw(k);
1291
1292 assertEquals(previous, map.put(k, v),
1293 "put must return the value that getRaw reported a moment ago, step " + step);
1294 assertEquals(v, map.getRaw(k), "the key just put is missing, step " + step);
1295
1296 expectedValues.put(k, v);
1297 mustBeAbsent.remove(k);
1298 }
1299 case Op.Get(String k) ->
1300 {
1301 final var actual = map.get(k);
1302
1303 if (mustBeAbsent.contains(k))
1304 {
1305 assertNull(actual, "get found " + k + " although it is not in the map, step " + step);
1306 }
1307 else if (actual != null)
1308 {
1309 assertEquals(expectedValues.get(k), actual, "get returned a wrong value for " + k);
1310 }
1311 }
1312 case Op.GetRaw(String k) ->
1313 {
1314 final var actual = map.getRaw(k);
1315
1316 if (mustBeAbsent.contains(k))
1317 {
1318 assertNull(actual, "getRaw found " + k + " although it is not in the map, step " + step);
1319 }
1320 else if (actual != null)
1321 {
1322 assertEquals(expectedValues.get(k), actual, "getRaw returned a wrong value for " + k);
1323 }
1324 }
1325 case Op.Remove(String k) ->
1326 {
1327 final var previous = map.getRaw(k);
1328
1329 assertEquals(previous, map.remove(k),
1330 "remove must return the value that was stored, step " + step);
1331 assertNull(map.getRaw(k), "the key is still there after remove: " + k + ", step " + step);
1332
1333 expectedValues.remove(k);
1334 mustBeAbsent.add(k);
1335 }
1336 case Op.Clear() ->
1337 {
1338 map.clear();
1339
1340 assertEquals(0, map.size(), "size after clear, step " + step);
1341 assertTrue(map.keys().isEmpty(), "keys after clear, step " + step);
1342
1343 expectedValues.clear();
1344 mustBeAbsent.clear();
1345
1346 for (int i = 0; i < keySpace; i++)
1347 {
1348 mustBeAbsent.add(key(i));
1349 }
1350 }
1351 }
1352
1353 verifyInvariants(map, maxSize, expectedValues, mustBeAbsent, step);
1354 }
1355 }
1356
1357 /**
1358 * @param random the source of randomness
1359 * @param keySpace how many distinct keys may be used
1360 * @param step the current step, used to build unique values
1361 * @return the next operation to apply
1362 */
1363 private Op nextOperation(final Random random, final int keySpace, final int step)
1364 {
1365 final var k = key(random.nextInt(keySpace));
1366 final int roll = random.nextInt(100);
1367
1368 if (roll < 50)
1369 {
1370 return new Op.Put(k, "v-" + step);
1371 }
1372 if (roll < 70)
1373 {
1374 return new Op.Get(k);
1375 }
1376 if (roll < 85)
1377 {
1378 return new Op.GetRaw(k);
1379 }
1380 if (roll < 99)
1381 {
1382 return new Op.Remove(k);
1383 }
1384
1385 return new Op.Clear();
1386 }
1387
1388 /**
1389 * Checks everything that must be true after every single operation. Only
1390 * LRU neutral methods are used here so the check itself cannot influence
1391 * what the map does next.
1392 *
1393 * @param map the map under test
1394 * @param maxSize its maximum size
1395 * @param expectedValues the value each known key must carry if present
1396 * @param mustBeAbsent keys that may not be in the map
1397 * @param step the step number, for the failure message
1398 */
1399 private void verifyInvariants(final LRUClockMap<String, String> map,
1400 final int maxSize,
1401 final Map<String, String> expectedValues,
1402 final Set<String> mustBeAbsent,
1403 final int step)
1404 {
1405 final int size = map.size();
1406 final var keys = map.keys();
1407
1408 assertTrue(size >= 0, "negative size in step " + step);
1409 assertTrue(size <= maxSize, "size " + size + " exceeds maxSize " + maxSize + " in step " + step);
1410 assertNotNull(keys, "keys() returned null in step " + step);
1411 assertEquals(size, keys.size(), "keys() and size() disagree in step " + step);
1412 assertEquals(size, Set.copyOf(keys).size(), "keys() contains duplicates in step " + step);
1413
1414 for (final var k : keys)
1415 {
1416 assertNotNull(k, "keys() contains null in step " + step);
1417 assertEquals(expectedValues.get(k), map.getRaw(k), "wrong value for " + k + " in step " + step);
1418 assertFalse(mustBeAbsent.contains(k), "removed key " + k + " came back in step " + step);
1419 }
1420
1421 for (final var k : mustBeAbsent)
1422 {
1423 assertNull(map.getRaw(k), "key " + k + " must not be in the map in step " + step);
1424 }
1425 }
1426 }

1497 * A key with a fully controllable hash code. Two keys are equal only when
1498 * both components match, so equal keys always report equal hash codes and
1499 * the {@code equals}/{@code hashCode} contract holds.
1500 *
1501 * @param id distinguishes keys that are meant to collide
1502 * @param hash the hash code this key reports
1503 */
1504 record HashedKey(int id, int hash)
1505 {
1506 /** The hash code shared by every key built with {@link #colliding(int)}. */
1507 private static final int SHARED_HASH = 0x5EED;
1508
1509 @Override
1510 public int hashCode()
1511 {
1512 return hash;
1513 }
1514
1515 /**
1516 * @param id the distinguishing id
1517 * @return a key that collides with every other key built this way
1518 */
1519 static HashedKey colliding(final int id)
1520 {
1521 return new HashedKey(id, SHARED_HASH);
1522 }
1523 }
